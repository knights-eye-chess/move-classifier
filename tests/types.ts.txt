import { recordDecision, readRecordedDecision, type RecordedDecision, type PolicyIdentity } from '@knights-eye-chess/move-classifier/contracts';
const identity: PolicyIdentity = { id: 'example/policy', revision: 1 };
declare const raw: unknown;
const snapshot: RecordedDecision = recordDecision(raw);
// @ts-expect-error Stored occurrence history is immutable.
snapshot.occurrence.history.push('e2e4');
const restored = readRecordedDecision(snapshot, identity);
const evidence: false = restored.reusableAsEngineEvidence;
void evidence;
// @ts-expect-error A numeric classifier revision alone is not a policy identity.

import { createKnightsEyePolicy, type KnightsEyeDecision } from '@knights-eye-chess/move-classifier/policy/knights-eye';
const ke = createChessAnalyzerCore(Chess, undefined, createKnightsEyePolicy(Chess));
const outcome = await ke.classifySuppliedEvidence({ startFen: new Chess().fen(), history: ['e2e4'], depth: 18, evidence: [] });
if (outcome.status === 'complete') {
  const decision: KnightsEyeDecision = outcome.decision;
  const loss: number = decision.qualityLoss;
  void loss;
} else {
  const reason: string = outcome.reason;
  void reason;
  // @ts-expect-error Unavailable outcomes cannot be treated as classifications.
  void outcome.decision;
}
